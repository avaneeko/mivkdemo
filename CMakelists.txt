cmake_minimum_required(VERSION 3.21)
project(camellia C CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# --- Dependencies ---------------------------------------------------------

# Vulkan SDK: found via the VULKAN_SDK env var set by the official installer.
find_package(Vulkan REQUIRED)

include(FetchContent)

FetchContent_Declare(cglm
    GIT_REPOSITORY https://github.com/recp/cglm.git
    GIT_TAG        v0.9.6
    GIT_SHALLOW    TRUE
)

FetchContent_Declare(fastgltf
    GIT_REPOSITORY https://github.com/spnda/fastgltf.git
    GIT_TAG        v0.8.0
    GIT_SHALLOW    TRUE
)

# Build cglm as a static lib, skip its tests/docs.
set(CGLM_STATIC      ON  CACHE BOOL "" FORCE)
set(CGLM_SHARED      OFF CACHE BOOL "" FORCE)
set(CGLM_USE_TEST    OFF CACHE BOOL "" FORCE)

FetchContent_MakeAvailable(cglm fastgltf)

# --- Target ---------------------------------------------------------------

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/src/*.c"
    "${CMAKE_SOURCE_DIR}/src/*.cpp"
)

add_executable(camellia ${SOURCES})

target_include_directories(camellia PRIVATE src/Core)

target_link_libraries(camellia PRIVATE
    Vulkan::Vulkan
    fastgltf::fastgltf
    cglm
)

# --- Defines --------------------------------------------------------------

target_compile_definitions(camellia PRIVATE
    _CRT_SECURE_NO_WARNINGS
    CGLM_FORCE_LEFT_HANDED
)

# --- Disable C++ exceptions ----------------------------------------------

if(MSVC OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(camellia PRIVATE
        $<$<COMPILE_LANGUAGE:CXX>:/EHs-c->
    )
    target_compile_definitions(camellia PRIVATE
        _HAS_EXCEPTIONS=0
        NOMINMAX=1
    )
else()
    target_compile_options(camellia PRIVATE
        $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
    )
endif()

# --- Windows system libraries --------------------------------------------

if(WIN32)
    target_link_libraries(camellia PRIVATE user32 gdi32)
endif()

# --- Per-mode flags -------------------------------------------------------

if(MSVC)
    target_compile_options(camellia PRIVATE
        $<$<CONFIG:Debug>:/Od /Zi>
        $<$<CONFIG:Release>:/O2>
    )
    target_link_options(camellia PRIVATE
        $<$<CONFIG:Release>:/OPT:REF /OPT:ICF>
    )
else()
    target_compile_options(camellia PRIVATE
        $<$<CONFIG:Debug>:-O0 -g>
        $<$<CONFIG:Release>:-O3 -fvisibility=hidden>
    )
    target_link_options(camellia PRIVATE
        $<$<CONFIG:Release>:-s>
    )
endif()
